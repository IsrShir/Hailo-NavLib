# ROS-free build of OpenVINS (ov_core + ov_init + ov_msckf, no ROS / no aruco) plus the EuRoC mono replay runner
cmake_minimum_required(VERSION 3.10)
project(ov_noros CXX)
set(OV_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../open_vins" CACHE PATH "path to open_vins source root (default: the VIO/openvins/open_vins submodule)")
set(CMAKE_CXX_STANDARD 14)
if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif ()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -g -DNDEBUG -fno-signed-zeros -fno-math-errno -funroll-loops" CACHE STRING "")
find_package(Eigen3 REQUIRED)
option(OV_OPENCV_MANUAL "link OpenCV modules directly (Yocto SDK OpenCVConfig is incomplete)" OFF)
set(OV_CV_MODULES core imgproc imgcodecs features2d video calib3d highgui)
if (OV_OPENCV_MANUAL)
  find_path(OpenCV_INCLUDE_DIRS opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
  set(OpenCV_LIBRARIES "")
  foreach (m ${OV_CV_MODULES})
    find_library(CV_${m} opencv_${m} REQUIRED)
    list(APPEND OpenCV_LIBRARIES ${CV_${m}})
  endforeach ()
else ()
  find_package(OpenCV REQUIRED COMPONENTS ${OV_CV_MODULES})
endif ()
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)
find_package(Ceres REQUIRED)
add_definitions(-DENABLE_ARUCO_TAGS=0 -DROS_AVAILABLE=0)
file(GLOB_RECURSE CORE ${OV_SRC}/ov_core/src/*.cpp)
list(FILTER CORE EXCLUDE REGEX ".*/test_[^/]*\\.cpp$")
file(GLOB_RECURSE INIT ${OV_SRC}/ov_init/src/*.cpp)
list(FILTER INIT EXCLUDE REGEX ".*/test_[^/]*\\.cpp$")
list(FILTER INIT EXCLUDE REGEX ".*Simulator\\.cpp$")
set(MSCKF dummy.cpp sim/Simulator.cpp state/State.cpp state/StateHelper.cpp state/Propagator.cpp core/VioManager.cpp
    core/VioManagerHelper.cpp update/UpdaterHelper.cpp update/UpdaterMSCKF.cpp update/UpdaterSLAM.cpp update/UpdaterZeroVelocity.cpp)
list(TRANSFORM MSCKF PREPEND ${OV_SRC}/ov_msckf/src/)
add_library(ov_lib STATIC ${CORE} ${INIT} ${MSCKF})
target_include_directories(ov_lib PUBLIC ${OV_SRC}/ov_core/src ${OV_SRC}/ov_init/src ${OV_SRC}/ov_msckf/src
                           ${EIGEN3_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
target_link_libraries(ov_lib PUBLIC ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} ${CERES_LIBRARIES})
add_executable(vio_replay vio_replay.cpp)
target_link_libraries(vio_replay ov_lib)
